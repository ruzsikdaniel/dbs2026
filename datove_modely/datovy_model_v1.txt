Table airport {
  id integer [primary key] 
  name varchar
  code varchar
}

Table passenger {
  id integer [primary key]
  firstname varchar
  lastname varchar
}

Table ticket {
  id integer [primary key]
  flight_id integer
  passenger_id integer
  price float
  seat_number integer

}

Table flight {
  id integer [primary key]
  airplane_id integer
  dest_airport_id integer
  src_airport_id integer
  runway_id integer
  departure_time timestamp
  arrival_time timestamp
}

Table airplane {
  id integer [primary key]
  airline varchar
  status varchar
}

Table baggage {
  id integer [primary key]
  passenger_id integer
  weight float
}
Table runway {
  id integer [primary key]
  airport_id integer
  name varchar
  status varchar
}

Table airplane_hangar {
  airplane_id integer [primary key]
  hangar_id integer [primary key]
}

Table hangar {
  id integer [primary key]
  airport_id integer
  name varchar
  capacity integer
  status varchar
}
